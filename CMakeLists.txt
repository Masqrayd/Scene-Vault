cmake_minimum_required(VERSION 3.28...3.30)

# Windows uses the official OBS template's dependency download and packaging.
option(SCENE_VAULT_SYSTEM_BUILD "Use installed OBS and Qt packages (Linux test builds)" OFF)
if(NOT SCENE_VAULT_SYSTEM_BUILD)
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
endif()
project(scene-vault VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT SCENE_VAULT_SYSTEM_BUILD)
  set(ENABLE_FRONTEND_API ON CACHE BOOL "OBS frontend required" FORCE)
  set(ENABLE_QT ON CACHE BOOL "Qt required" FORCE)
  include(compilerconfig)
  include(defaults)
  include(helpers)
endif()

find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
add_library(scene-vault MODULE src/plugin-main.cpp src/scene-vault.cpp src/scene-controller.cpp)
target_link_libraries(scene-vault PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets)
if(SCENE_VAULT_SYSTEM_BUILD)
  set_target_properties(scene-vault PROPERTIES PREFIX "")
else()
  set_target_properties_plugin(scene-vault PROPERTIES OUTPUT_NAME "scene-vault")
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(controller-behavior-tests tests/controller-behavior-tests.cpp src/scene-controller.cpp)
  target_link_libraries(controller-behavior-tests PRIVATE OBS::libobs Qt6::Core Qt6::Widgets Qt6::Test)
  target_include_directories(controller-behavior-tests PRIVATE src $<TARGET_PROPERTY:OBS::obs-frontend-api,INTERFACE_INCLUDE_DIRECTORIES>)
  add_test(NAME controller-behavior COMMAND controller-behavior-tests)
  set_tests_properties(controller-behavior PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(vault-state-tests tests/vault-state-tests.cpp)
  target_link_libraries(vault-state-tests PRIVATE Qt6::Core)
  add_test(NAME vault-state COMMAND vault-state-tests)
  add_executable(vault-integration-tests tests/vault-integration-tests.cpp src/scene-vault.cpp src/scene-controller.cpp)
  target_link_libraries(vault-integration-tests PRIVATE OBS::libobs Qt6::Core Qt6::Widgets Qt6::Test)
  target_include_directories(vault-integration-tests PRIVATE src)
  target_include_directories(vault-integration-tests PRIVATE $<TARGET_PROPERTY:OBS::obs-frontend-api,INTERFACE_INCLUDE_DIRECTORIES>)
  add_test(NAME vault-integration COMMAND vault-integration-tests)
  set_tests_properties(vault-integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
